\documentclass[tikz,border=0pt]{standalone}
\usepackage[T1]{fontenc}
\usepackage[sfdefault]{FiraSans}

\usetikzlibrary{positioning,fit,backgrounds,arrows.meta,calc,shadows}

\definecolor{dataC}{HTML}{2563EB}
\definecolor{modelC}{HTML}{EA580C}

% \card{id}{x,y}{colour}{title}{subtitle}
\newcommand{\card}[5]{%
  \node[card=#3] (#1) at (#2) {{\large\bfseries #4}\\[1pt]{\normalsize\color{muted}#5}};}
% \badges{card id}{colour}{number of icons}{icon1,icon2,...}  - white logo circles on the card's top edge
\newcommand{\badges}[4]{%
  \foreach \ic [count=\j] in {#4} {
    \node[badge=#2] (#1-b\j) at ($(#1.north)+({(\j-(#3+1)/2)*1.4},0)$)  % 14 mm apart
      {\includegraphics[width=6.2mm,height=6.2mm,keepaspectratio]{icons/\ic}};
  }}
% \step{colour}{n} - white numbered circle for panel headers
\newcommand{\stepnum}[2]{\tikz[baseline=(s.base)]\node[circle, fill=white, text=#1, inner sep=0pt,
  minimum size=5.5mm, font=\normalsize\bfseries] (s) {#2};}

\begin{document}
\begin{tikzpicture}[
    card/.style={draw=#1, fill=white, line width=1.4pt, rounded corners=6pt,
                 minimum width=46mm, minimum height=21mm, text width=42mm,
                 align=center, inner ysep=5.5mm, text=ink,   % room for the logo badges
                 drop shadow={opacity=0.12, shadow xshift=0.7mm, shadow yshift=-0.9mm}},
    keycard/.style={card=#1, fill=#1!10, line width=2pt},
    badge/.style={circle, draw=#1, fill=white, line width=1.2pt, minimum size=10mm, inner sep=0pt},
    flow/.style={-{Stealth[length=3.4mm, width=2.8mm]}, line width=1.6pt, draw=line, rounded corners=5pt},
    ask/.style={flow, draw=askC, line width=2.4pt},
    panel/.style={fill=#1!5, draw=#1!35, line width=1pt, rounded corners=12pt,
                  inner xsep=6mm, inner ysep=13mm},
    header/.style={fill=#1, text=white, rounded corners=5pt, font=\large\bfseries,
                   inner xsep=3.5mm, inner ysep=2mm, anchor=west},
]

% ------------------------------------------------------------- cards
% (1) data pipeline - left column, flowing down
\card{src}{0,  5.6}{dataC}{CNCF Landscape}{docs, READMEs, PDFs}
\card{scrape}{0, 2.0}{dataC}{Scraping \& chunking}{Scrapy, Python}
\node[keycard=dataC] (corpus) at (0,-1.6) {{\large\bfseries CNCF corpus}\\[1pt]{\normalsize\color{muted}Hugging Face dataset}};

% (2) model - top row (spread over the same width as the RAG row)
\card{qa}{6.9, 5.6}{modelC}{Q\&A generation}{Gemma}
\card{ft}{15.45, 5.6}{modelC}{LoRA fine-tuning}{Gemma 2B/7B/9B}
\card{serve}{24.0, 5.6}{modelC}{Quantize \& serve}{GGUF, Ollama / LocalAI}

% (3) retrieval - bottom row
\card{emb}{6.9,-1.6}{ragC}{Embedding}{bge-small}
\card{index}{12.6,-1.6}{ragC}{Vector index}{Chroma}
\card{retr}{18.3,-1.6}{ragC}{Retriever}{top-$k$ relevant chunks}
\card{prompt}{24.0,-1.6}{ragC}{Prompt}{question + chunks}

% (4) DeepCNCF: the one model that writes the answer - fed by fine-tuning AND retrieval
\node[keycard=modelC, minimum width=52mm, text width=48mm, minimum height=30mm, line width=2.6pt] (deep) at (30.4,2.0)
  {{\rule{0pt}{6.5mm}\Large\bfseries DeepCNCF}\\[2pt]{\normalsize\color{muted}fine-tuned model\\generates the answer}};

% user
\node[card=askC, minimum width=40mm, text width=36mm, minimum height=14mm] (user) at (18.3,-6.6)
  {{\large\bfseries User}};
\node[keycard=askC, minimum height=14mm] (answer) at (30.4,-6.6)
  {{\large\bfseries Answer}\\[1pt]{\normalsize\color{muted}with sources}};

% ------------------------------------------------------------- flows
\draw[flow] (src) -- (scrape);
\draw[flow] (scrape) -- (corpus-b1);  % end on the logo badge, not under it
\draw[flow] (corpus.east) -- ++(12mm,0) |- (qa.west);
\draw[flow] (corpus) -- (emb);
\draw[flow] (qa) -- (ft);
\draw[flow] (ft) -- (serve);

% question path (highlighted): question -> retriever & prompt -> DeepCNCF -> answer -> user
\coordinate (fork) at ($(user.north)+(0,1.3)$);
\draw[ask] (user.north) -- (retr.south);
\draw[ask] (fork) -| (prompt.south);
\fill[askC] (fork) circle (1.6mm);
\node[font=\normalsize\bfseries, text=askC, left] at ($(fork)+(-1mm,0)$) {question};
\draw[ask] (retr) -- (prompt);
\draw[ask] (prompt.east) -- ++(9mm,0) |- (deep.west);
\draw[ask] (deep) -- (answer);
\draw[ask] (answer) -- (user);
